🛡️ Bash Options and Explicit Checks
#!/bin/bash
set -e # Exit on error
set -u # Error on undefined var
set -o pipefail # Rightmost nonzero pipeline status, or zero

# Or compact:
set -euo pipefail
⚠️ set -e has test-context exceptions: if/while/until, non-final &&/|| commands and tested functions. A final failure may exit. (...) inherits it; $(...) normally clears it unless inherit_errexit is enabled.
📦 Functions - Common Mistakes
# ❌ WRONG: global variable
process() { count=0; }

# ✅ CORRECT: local variable
process() { local count=0; }

# ❌ WRONG: return for values
get_sum() { return $((a+b)); } # Max 255!

# ✅ CORRECT: echo for values
get_sum() { echo $((a+b)); }
result=$(get_sum)

🔑 Associative Arrays
# MANDATORY declare -A !
declare -A config
config[host]="localhost"
config[port]="8080"

# Or inline:
declare -A config=(
[host]="localhost"
[port]="8080"
)
OperationSyntax
Value${config[host]}
All keys${!config[@]}
All values${config[@]}
Key count${#config[@]}
⚠️ WITHOUT declare -A: hash[host]="x" → hash[0]="x" (unset arithmetic host becomes 0 without nounset; it is not a string key)
🧹 Trap and Cleanup
TEMP="" # initialise before installing the trap
cleanup() {
local exit_code=$?
[[ -f "$TEMP" ]] && rm -f "$TEMP"
exit $exit_code
}

trap cleanup EXIT # Best-effort cleanup
trap 'exit 130' INT; trap 'exit 143' TERM # On Ctrl+C
Trap event / signalWhen
EXITOn ordinary shell exit; not SIGKILL/power loss
ERROn eligible nonzero status; -e is not required
INTCtrl+C
TERMkill (default)
DEBUGBefore each command
🔧 Default Values
SyntaxEffect
${VAR:-default}Return default if unset/empty
${VAR:=default}Set VAR to default if unset/empty
${VAR:?error}Error if unset/empty
${VAR:+alt}Return alt only if VAR is set and nonempty; VAR= gives an empty expansion
# Pattern for arguments:
INPUT="${1:-default.txt}"
: "${REQUIRED:?Must be set}"

📋 Professional Script Template
#!/bin/bash
set -euo pipefail

readonly SCRIPT_NAME=$(basename "$0")
die() { echo "FATAL: $*" >&2; exit 1; }

cleanup() {
local ec=$?
# cleanup code
exit $ec
}
trap cleanup EXIT

[[ $# -ge 1 ]] || die "Usage: $SCRIPT_NAME <arg>"

main() {
echo "Hello from $SCRIPT_NAME"
}

main "$@"
