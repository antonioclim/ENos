find - Search by name

Find paths by name using glob patterns.

# Case-sensitive
find . -name "*.txt"

# Case-insensitive
find . -iname "readme*"

# Exact match
find /etc -name "passwd"
🔍

find - Search by type

Filter results by file type.

FlagTip
-type fRegular file
-type dDirector
-type lSymbolic link
# Directories only
find . -type d

# Files .sh
find . -type f -name "*.sh"
🔍

find - Search by size

Find large or small files.

# Larger than 100 MiB
find . -size +100M

# Empty files: -size -1k rounds non-zero sizes up
find . -size -1k

# Empty files
find . -size 0 -type f

Suffixes: c=bytes, k=KiB (1024 bytes), M=MiB, G=GiB

🔍

find - Search by time

Find recently modified or older files.

# Within the last 24 hours
find . -mtime -1

# At least 31 complete 24-hour periods old
find . -mtime +30

# Within the last 60 minutes
find . -mmin -60

# Newer than ref.txt
find . -newer ref.txt
-mtime -N = less than N complete 24-hour periods old
-mtime +N = at least N+1 complete 24-hour periods old
🔍

find - Logical Operators

Combine conditions in a search expression.

# AND (implicit)
find . -type f -name "*.txt"

# OR
find . -name "*.txt" -o -name "*.md"

# NOT
find . ! -name "*.git"

# Grouping
find . \( -name "*.txt" -o -name "*.md" \)
🔍

find - Actions (-exec)

Execute a command for selected paths.

# For each selected file
find . -name "*.txt" -exec ls -l {} \;

# Batch (fewer command invocations)
find . -name "*.txt" -exec wc -l {} +

# With confirmation
find . -name "*.tmp" -ok rm {} \;
\; = one command per file
+ = bounded batches of arguments
⚠️

find -delete - CAUTION!

DANGER: Order matters! 
# ❌ WRONG - attempts to delete every visited path!
find . -delete -name "*.tmp"

# ✅ CORRECT
find . -name "*.tmp" -delete
Best practice: Preview the same predicate first with -print!
🔍

find - Depth limits

Control the search depth.

# Only in the current directory
find . -maxdepth 1 -type f

# At most two levels
find . -maxdepth 2 -name "*.txt"

# Exclude the starting point
find . -mindepth 1 -type d
🔗

xargs - Batch Processing

Construct command arguments from standard input.

# Count lines
find . -name "*.txt" | xargs wc -l

# Search in files
find . -name "*.py" | xargs grep "import"

# Delete files
find . -name "*.tmp" | xargs rm
🔗

xargs - Handling Spaces

Default xargs parsing does not preserve arbitrary filenames. 
# ❌ WRONG for filenames containing spaces
find . -name "*.txt" | xargs rm

# ✅ CORRECT - NUL delimiter
find . -name "*.txt" -print0 | xargs -0 rm
-print0 + -0 = preserves filename boundaries (still validate the selected paths)
🔗

xargs - Placeholder (-I)

Control where each argument is inserted.

# Preserve relative directories; exclude the destination from the traversal.
mkdir -p -- backup
find . -path ./backup -prune -o -type f -name '*.txt' -print0 |
    xargs -0 -r -I{} sh -c 'mkdir -p -- "backup/$(dirname -- "$1")" && cp -- "$1" "backup/$1"' sh {}
🔗

xargs - Parallel execution

Run bounded concurrent jobs.

# up to 4 concurrent processes
find . -name "*.log" | xargs -n 1 -P 4 gzip

# At most two arguments per process
find . -type f | xargs -n 2 -P 4 echo

# Image processing
ls *.jpg | xargs -P 8 -I{} convert {} -resize 50% thumb_{}

$@ vs $* - The difference

They treat argument boundaries differently. 
# Invocation: ./s.sh "a b" c

# "$@" = 2 elements
for arg in "$@"; do echo "[$arg]"; done
# [a b] [c]

# "$*" = 1 string
for arg in "$*"; do echo "[$arg]"; done
# [a b c]
Use always "$@"!
📜

shift - Shifting arguments

Remove the first argument and shift the remainder left.

# ./s.sh a b c d
echo $1  # a
shift
echo $1  # b
shift 2
echo $1  # d

# Process all arguments
while [[ $# -gt 0 ]]; do
    echo "Processing: $1"
    shift
done
📜

getopts - Option Parsing

Parse short options such as -a and -b value.

while getopts ":hvo:" opt; do
    case $opt in
        h) show_help ;;
        v) verbose=1 ;;
        o) output="$OPTARG" ;;
        :|\?) echo "Invalid option or missing argument: -$OPTARG" >&2; exit 2 ;;
    esac
done
shift $((OPTIND - 1))
: after a letter = requires an argument
OPTARG = argument value
📜

Default Values

Provide a value when a variable is unset or empty.

# Use a default (without modifying VAR)
echo "${VAR:-default}"

# Assign and use a default
echo "${VAR:=default}"

# Alternative value when set and non-empty
echo "${VAR:+alternative}"

# Error when unset or empty
echo "${VAR:?Error: VAR missing}"

Symbolic chmod

WHO: u/g/o/a | OP: +/-/= | PERM: r/w/x

# Add execute for owner
chmod u+x script.sh

# Remove write for group/others
chmod go-w file.txt

# Set exactly
chmod u=rwx,g=rx,o=r file

# Add read for all
chmod a+r file.txt
⚠️

chmod X vs x (Recursive)

chmod -R a+x makes every selected file executable! 
# ❌ GREȘIT
chmod -R a+x myproject/

# ✅ CORRECT - X = directories or files already executable by someone
chmod -R a+X myproject/

# ✅ ALTERNATIVE - find
find . -type d -exec chmod 755 {} \;
find . -type f -exec chmod 644 {} \;
🎭

chown / chgrp

# Change owner
sudo chown user file.txt

# Change owner:group
sudo chown user:group file.txt

# Group only
chgrp developers file.txt
sudo chown :developers file.txt

# Recursive
sudo chown -R www-data:www-data /var/www/

Cron - Common examples

# Every 5 minutes
*/5 * * * * /path/script.sh

# Daily at 03:00
0 3 * * * /path/backup.sh

# Monday–Friday at 09:00
0 9 * * 1-5 /path/work.sh

# First day of the month
0 0 1 * * /path/monthly.sh

# @shortcuts
@reboot    /path/startup.sh
@daily     /path/daily.sh
@hourly    /path/hourly.sh
⚠️

Cron - Common pitfalls

Cron runs with a MINIMAL! 
# ❌ GREȘIT - python may not be in PATH
* * * * * python script.py

# ✅ CORRECT - absolute path
* * * * * /usr/bin/python3 /home/user/script.py

# ✅ With logging
0 3 * * * /path/backup.sh >> /home/user/backup.log 2>&1
DOM + DOW = OR, not AND!
0 9 15 * 1 = on the 15th OR on Mondays
