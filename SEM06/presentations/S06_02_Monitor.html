CPU Monitoring

get_cpu_usage() {
    local first second total1 idle1 total2 idle2 elapsed idle_elapsed
    [[ -r /proc/stat ]] || { log_error "CPU counters are unavailable"; return 1; }
    first=$(grep '^cpu ' /proc/stat) || return 1
    sleep "${CPU_SAMPLE_INTERVAL:-0.5}"
    second=$(grep '^cpu ' /proc/stat) || return 1
    # The first eight counters include steal. guest/guest_nice are already counted
    # in user/nice. This model treats idle+iowait as non-busy time.
    IFS=' ' read -r total1 idle1 < <(awk '{s=0; for(i=2;i<=9;i++) s+=$i; print s,$5+$6}' <<< "$first")
    IFS=' ' read -r total2 idle2 < <(awk '{s=0; for(i=2;i<=9;i++) s+=$i; print s,$5+$6}' <<< "$second")
    elapsed=$((total2-total1)); idle_elapsed=$((idle2-idle1))
    (( elapsed > 0 && idle_elapsed >= 0 && idle_elapsed <= elapsed )) || {
        log_error "CPU counters did not yield a valid interval"; return 1;
    }
    printf '%s\n' "$((100 * (elapsed-idle_elapsed) / elapsed))"
}

CPU - Per Core Details

get_cpu_per_core() (
    local snapshots
    snapshots=$(mktemp -d) || return 1
    trap 'rm -rf -- "$snapshots"' EXIT
    awk '/^cpu[0-9]+ /' /proc/stat > "$snapshots/before" || return 1
    sleep 1 || return 1
    awk '/^cpu[0-9]+ /' /proc/stat > "$snapshots/after" || return 1
    awk '
    function total( i,s) {s=0; for(i=2;i<=9;i++) s+=$i; return s}
    NR==FNR {previous[$1]=total(); idle[$1]=$5+$6; next}
    {delta=total()-previous[$1]; di=$5+$6-idle[$1]
     if(!($1 in previous) || delta<=0 || di<0 || di>delta)
         printf "%s unavailable\n",$1
     else printf "%s %.2f%%\n",$1,100*(delta-di)/delta}
    ' "$snapshots/before" "$snapshots/after"
)
# Two interval samples, including steal; guest counters are not added twice.
# Core removal/addition or a counter reset yields unavailable, not 0% idle.

Memory Monitoring

get_memory_info() {
    awk '
    /^MemTotal:/ {total=$2}
    /^MemAvailable:/ {available=$2; found=1}
    /^MemFree:/ {free=$2}
    /^Buffers:/ {buffers=$2}
    /^Cached:/ {cached=$2}
    END {
        if(!found || total<=0 || available<0 || available>total) exit 1
        printf "total_kib:%s\nused_kib:%s\nfree_kib:%s\n",total,total-available,free
        printf "available_kib:%s\nbuffers_kib:%s\ncached_kib:%s\n",available,buffers,cached
        printf "usage_percent:%.2f\n",100*(total-available)/total
    }' /proc/meminfo
}
get_memory_usage() {
    local info
    info=$(get_memory_info) || return 1
    awk -F: '$1=="usage_percent" {print $2}' <<< "$info"
}
# Linux /proc/meminfo labels kB but these fields represent KiB.
# MemAvailable is an estimate; it is not the same quantity as MemFree.

Disk Monitoring

get_disk_usage() {
    local mount_point=${1:-/} output device total used available percent mount
    output=$(LC_ALL=C df -Pk -- "$mount_point") || return 1
    IFS=' ' read -r device total used available percent mount <<< "${output##*$'\n'}"
    percent=${percent%\%}
    [[ $total =~ ^[0-9]+$ && $percent =~ ^[0-9]+$ ]] || return 1
    printf 'device:%s\ntotal_kib:%s\nused_kib:%s\navailable_kib:%s\nusage_percent:%s\nmount_point:%s\n' \
        "$device" "$total" "$used" "$available" "$percent" "$mount"
}
get_disk_io() {
    local device=${1:-sda} reads read_sectors writes write_sectors
    [[ $device =~ ^[A-Za-z0-9._-]+$ && -r /sys/block/$device/stat ]] || return 1
    IFS=' ' read -r reads _ read_sectors _ writes _ write_sectors _ < "/sys/block/$device/stat" || return 1
    printf 'reads:%s writes:%s read_sectors:%s write_sectors:%s\n' "$reads" "$writes" "$read_sectors" "$write_sectors"
}
# /sys/block statistics are cumulative; subtract two snapshots to estimate a rate.

Process Monitoring

get_top_processes() {
    local count=${1:-5} sort_by=${2:-cpu} field
    [[ $count =~ ^[1-9][0-9]*$ && ${#count} -lt 6 ]] || return 2
    case $sort_by in cpu) field=pcpu;; memory|mem) field=pmem;; *) return 2;; esac
    # Explicit columns; sed reads the full stream, avoiding head/SIGPIPE under pipefail.
    LC_ALL=C ps -eo pid=,user=,pcpu=,pmem=,args= --sort="-$field" | sed -n "1,${count}p"
}
monitor_process() {
    python3 - "$1" <<'PYPROC'
import json, pathlib, sys
try:
    pid=int(sys.argv[1]); assert pid>0
    raw=pathlib.Path(f'/proc/{pid}/stat').read_text()
    fields=raw[raw.rfind(')')+2:].split()  # comm may contain spaces or parentheses
    print(json.dumps({'pid':pid,'state':fields[0], 'ppid':int(fields[1]),
        'start_ticks':int(fields[19]), 'vsize_bytes':int(fields[20]), 'rss_pages':int(fields[21])}))
except (OSError, ValueError, IndexError, AssertionError) as error:
    print(f'Process unavailable: {error}',file=sys.stderr); raise SystemExit(1)
PYPROC
}
# PID plus start_ticks distinguishes PID reuse; one observation is not a leak detector.

Alerting System

declare -A THRESHOLDS=([cpu_warning]=70 [cpu_critical]=90
    [memory_warning]=75 [memory_critical]=95 [disk_warning]=80 [disk_critical]=95)
check_thresholds() {
    local metric=$1 value=$2
    [[ $metric == cpu || $metric == memory || $metric == disk ]] || return 3
    python3 - "$value" "${THRESHOLDS[${metric}_warning]}" "${THRESHOLDS[${metric}_critical]}" <<'PYALERT'
import math,sys
try:
    value,warning,critical=map(float,sys.argv[1:])
    if not all(map(math.isfinite,(value,warning,critical))) or not 0<=warning<=critical<=100 or not 0<=value<=100:
        raise ValueError('Invalid threshold/measurement')
    status=2 if value>=critical else 1 if value>=warning else 0
    print(('OK','WARNING','CRITICAL')[status]); raise SystemExit(status)
except ValueError:
    print('UNAVAILABLE'); raise SystemExit(3)
PYALERT
}
# Capture a nonzero alert status explicitly under set -e.
if status=$(check_thresholds cpu 95); then rc=0; else rc=$?; fi
printf 'Measurement: %s (status %s)\n' "$status" "$rc"
# These per-metric statuses are distinct from the CLI aggregate (0 healthy, 2 alert).

Format Implementation

format_output() {
    # One encoder consumes one set of measurements. Invalid/unavailable data is null.
    python3 - "${1:-text}" "$2" "$3" "$4" <<'PYFORMAT'
import csv,datetime,json,math,sys
fmt=sys.argv[1]; timestamp=datetime.datetime.now(datetime.timezone.utc).isoformat()
def metric(value):
    if value=='unavailable': return None
    number=float(value)
    if not math.isfinite(number) or not 0<=number<=100: raise ValueError('Invalid metric')
    return number
try:
    values=list(map(metric,sys.argv[2:])); names=('cpu','memory','disk')
    if fmt=='json': print(json.dumps({'timestamp':timestamp,**{n:{'usage':v} for n,v in zip(names,values)}},allow_nan=False))
    elif fmt=='csv': csv.writer(sys.stdout).writerow([timestamp,*values])
    elif fmt=='text': print(timestamp,*(f'{n}: {v if v is not None else "unavailable"}%' for n,v in zip(names,values)))
    elif fmt=='prometheus':
        for n,v in zip(names,values):
            if v is not None: print(f'system_{n}_usage_percent {v}')
    else: raise ValueError('Unknown format')
except ValueError as error:
    print(error,file=sys.stderr); raise SystemExit(2)
PYFORMAT
}

Main Monitoring Loop

monitor_continuous() {
    local program=${MONITOR_PROGRAM:?Set the absolute path to monitor.sh}
    # The delivered CLI owns metric collection, alert aggregation and cleanup.
    # --daemon accepts --interval; format is selected with --output.

Best Practices

Performance

  • Cache reads /proc
  • Avoid subshells
  • Batch multiple metrics
  • Adaptive sampling

Reliability

  • Validate files /proc
  • Fallback to alternatives
  • Operation timeout
  • Graceful degradation

Alerting

Hysteresis to avoid alert flapping | Cooldown between alerts | Aggregate similar alerts
