The 3-2-1 Rule

3

Copies of the data

Original + two backups

2

Different storage media

HDD + Cloud/Tape

1

Off-site copy

A separate geographical location

Separate media/locations reduce some correlated failures. Ransomware resilience also requires an access/isolation model and tested recovery.

create_full_backup() {
    bash "${BACKUP_PROGRAM:?Set the absolute path to backup.sh}" \
        --source "$1" --dest "$2" --type full --incremental --compression "${3:-gz}"
}
# --incremental with --type full creates the GNU tar level-0 snapshot.
# The core stages an archive and snapshot, verifies structure and checksum,
# then publishes .parent, .snapshot, .latest and .sources metadata.
# An explicit --type full does not run automatic rotation.

Incremental Backup

create_incremental_backup() {
    bash "${BACKUP_PROGRAM:?}" --source "$1" --dest "$2" \
        --type daily --incremental --compression "${3:-gz}"
}
# A matching .sources identity binds the source/exclusion set to the chain.
# GNU tar --listed-incremental stores directory state, including deletions.
# .parent names the exact predecessor; timestamp substrings never select a chain.
# Failure does not advance the checkpoint. A missing base starts a full chain.
# Concurrent live application writes still need an application-consistent snapshot.

Compression Implementation

compress_file() (
    local input=$1 algorithm=${2:-gzip} level=${3:-6} extension output stage before after ratio
    [[ -f $input && $level =~ ^[1-9]$|^1[0-9]$ ]] || return 2
    case $algorithm in
        gzip) extension=gz; ((level <= 9)) || return 2 ;;
        bzip2) extension=bz2; ((level <= 9)) || return 2 ;;
        xz) extension=xz; ((level <= 9)) || return 2 ;;
        zstd) extension=zst ;;
        *) return 2 ;;
    esac
    output=$input.$extension
    [[ ! -e $output && ! -L $output ]] || return 1
    stage=$(mktemp "$(dirname -- "$input")/.compress.XXXXXX") || return 1
    trap 'rm -f -- "$stage"' EXIT
    "$algorithm" -"$level" -c -- "$input" > "$stage" || return 1
    before=$(stat -c %s -- "$input") || return 1
    after=$(stat -c %s -- "$stage") || return 1
    if ((before == 0)); then ratio='undefined (empty input)'
    else ratio=$(awk -v a="$before" -v b="$after" 'BEGIN {printf "%.2f%%",100*(1-b/a)}'); fi
    mv -- "$stage" "$output" || return 1
    printf 'Bytes: %s -> %s; reduction: %s\n' "$before" "$after" "$ratio" >&2
    printf '%s\n' "$output"
)

Integrity Verification

checksum_tool() {
    case ${1:-sha256} in
        md5|sha1|sha256|sha512) printf '%ssum\n' "$1" ;;
        *) printf 'Unsupported checksum algorithm: %s\n' "$1" >&2; return 2 ;;
    esac
}
generate_checksum() (
    local file=$1 algorithm=${2:-sha256} tool directory base temporary
    tool=$(checksum_tool "$algorithm") || return 2
    directory=$(dirname -- "$file"); base=$(basename -- "$file")
    [[ -f $file && -r $file ]] || return 1
    temporary=$(mktemp "$directory/.checksum.XXXXXX") || return 1
    trap 'rm -f -- "$temporary"' EXIT
    (cd -- "$directory" && "$tool" -- "$base") > "$temporary" || return 1
    mv -- "$temporary" "$file.$algorithm"
)
verify_checksum() (
    local file=$1 algorithm=${2:-sha256} tool directory base
    tool=$(checksum_tool "$algorithm") || return 2
    directory=$(dirname -- "$file"); base=$(basename -- "$file")
    [[ -f $file && -r $file.$algorithm ]] || return 1
    # Recalculate the exact file; sidecars cannot redirect verification elsewhere.
    python3 - "$file" "$file.$algorithm" "$algorithm" <<'PY'
import hashlib, pathlib, re, sys
archive, sidecar, algorithm = sys.argv[1:]
try:
    expected = pathlib.Path(sidecar).read_text().split()[0].lstrip('\\')
    if not re.fullmatch('[0-9a-fA-F]+', expected): raise ValueError('invalid digest')
    with open(archive, 'rb') as stream:
        digest = hashlib.new(algorithm)
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(chunk)
        actual = digest.hexdigest()
    raise SystemExit(0 if actual == expected.lower() else 1)
except (OSError, ValueError, IndexError) as error:
    print(error, file=sys.stderr); raise SystemExit(1)
PY
)

Archive Verification

verify_archive() {
    local archive=$1 algorithm=${2:-sha256}
    [[ -f $archive && -r $archive ]] || return 1
    case $archive in
        *.tar|*.tar.gz|*.tgz|*.tar.bz2|*.tar.xz) tar -tf "$archive" >/dev/null || return 1 ;;
        *.tar.zst) tar --zstd -tf "$archive" >/dev/null || return 1 ;;
        *) return 2 ;;
    esac
    verify_checksum "$archive" "$algorithm" || return 1
    printf 'Archive structure and %s checksum verified: %s\n' "$algorithm" "$archive" >&2
}
# Missing/corrupt checksum and unknown algorithm fail explicitly.
# This four-algorithm teaching helper differs from the CLI's documented set.
# A checksum is integrity evidence, not publisher authentication.

Incremental Restore

restore_incremental_chain() {
    local target_archive=$1 destination=$2
    bash "${BACKUP_PROGRAM:?}" --dest "$(dirname -- "$target_archive")" \
        --restore "$target_archive" --restore-dir "$destination"
}
# Select an exact target archive, not a date guessed from its filename.
# Restore walks .parent back to the full base, rejects cycles/missing ancestors,
# and applies GNU tar increments in base-to-target order.
# Acceptance fixture: create, modify, delete and rename; compare the whole tree.

check_disk_space() {
    local dest=$1 required_mb=$2 available output
    [[ $required_mb =~ ^[0-9]+$ ]] || return 2
    output=$(LC_ALL=C df -Pm -- "$dest") || return 1
    available=$(awk 'NR==2 {print $4}' <<< "$output")
    [[ $available =~ ^[0-9]+$ ]] || return 1
    ((available >= required_mb))
}

Backup System - Enterprise backup system

USAGE:
    backup.sh [options] [sources...]

MAIN OPTIONS:
    -s, --source PATH       Add backup source (can be repeated)
    -d, --dest PATH         Destination directory for backups
    -t, --type TYPE         Backup type: auto, daily, weekly, monthly, full
    --compression TYPE      Compression: gz, bz2, xz, zstd, none (default: gz)

RETENTION:
    --retention-daily N     Keep N daily backups (default: 7)
    --retention-weekly N    Keep N weekly backups (default: 4)
    --retention-monthly N   Keep N monthly backups (default: 12)

EXCLUSIONS:
    -x, --exclude PATTERN   Exclude pattern from backup (can be repeated)
    --prefix NAME           Prefix for backup name (default: backup)

VERIFICATIONS:
    --incremental           GNU tar incremental chain (first run creates its full base)
    --list                  List stored archives
    --restore ARCHIVE       Restore an archive or its complete incremental chain
    --restore-dir PATH      Empty destination for a chain restore
    --verify-archive FILE   Check archive structure
    --no-verify             Don't verify backup integrity
    --no-checksum           Don't save checksum file

NOTIFICATIONS:
    -e, --email ADDRESS     Email address for notifications

LOGGING:
    -l, --log-file FILE     Log file
    --log-level LEVEL       Level: DEBUG, INFO, WARN, ERROR

GENERAL:
    -c, --config FILE       Configuration file
    -n, --dry-run           Simulation, don't actually execute
    -v, --verbose           Detailed output
    -h, --help              Display this message
    --version               Display version

EXAMPLES:
    # Simple backup
    backup.sh -s /home -d /backup
    
    # Backup with exclusions
    backup.sh -s /home -d /backup -x '*.tmp' -x 'cache'
    
    # Backup with notifications
    backup.sh -s /var/www -d /backup -e admin@example.com
    
    # Dry-run for testing
    backup.sh -s /data -d /backup -n -v

BACKUP TYPES:
    auto      - Auto-detect: monthly on 1st, weekly on Sunday, otherwise daily
    daily     - Daily backup, short retention
    weekly    - Weekly backup
    monthly   - Monthly backup, long retention
    full      - Full backup without rotation

CONFIGURATION FILE:
    Literal KEY=VALUE settings; no shell expansion or execution.
    Arrays use quoted (...) elements. CLI settings override the file.
    Default location: ./etc/backup.conf

EXIT CODES:
    0 - Success
    1 - Configuration error
    2 - Backup error
    3 - Verification error

Best Practices

Security

  • Encrypt sensitive data
  • Checksums for integrity
  • Restrictive permissions
  • Audit trail

Reliability

  • Test restoration periodically
  • Monitor execution
  • Report failures
  • Independent locations

Golden Rule

Treat recoverability as unverified until an appropriate restore test succeeds.
