Strategy Comparison

These are design trade-offs, not measured guarantees. Database/schema changes, session state, capacity, adapter behaviour and observation windows determine the actual risk.

Strategy Downtime Resources Rollback Risk
Rolling Conditional N instances Intermediate Intermediate
Blue-Green Conditional 2x instances Traffic reversal; state may differ Low
Canary Conditional N+few Traffic reversal; state may differ Very low
Recreate Yes N instances Slow High

Rolling - Implementation

deploy_rolling() {
    local app=$1 version=$2; shift 2
    local -a instances=("$@") batch=()
    local batch_size=${ROLLING_BATCH_SIZE:-1} delay=${ROLLING_DELAY:-0}
    local total=${#instances[@]} i instance deployed=0
    [[ $batch_size =~ ^[1-9][0-9]{0,5}$ && $delay =~ ^[0-9]{1,4}$ ]] || return 2
    ((total > 0)) || return 2
    for ((i=0; i<total; i+=batch_size)); do
        batch=("${instances[@]:i:batch_size}")
        for instance in "${batch[@]}"; do
            if drain_instance "$instance" && deploy_to_instance "$instance" "$app" "$version" &&
               wait_for_health "$instance" "${HEALTH_TIMEOUT:-30}" && enable_instance "$instance"; then
                deployed=$((deployed + 1))
            else
                if ! rollback_instance "$instance" "$app"; then return 3; fi
                return 1  # Stop even when rollback succeeds.
            fi
        done
        if ((i + batch_size < total)); then sleep "$delay" || return 1; fi
    done
    printf 'Rolling complete: %s/%s\n' "$deployed" "$total"
}

     T2: Test GREEN, smoke tests
     T3: Switch traffic BLUE → GREEN

✅ Traffic can switch back if the adapter supports it; data/state rollback is separate | 🎯 Routing boundary | ⚠️ 2x resources

Blue-Green - Implementation

deploy_blue_green() {
    local app=$1 version=$2 active target
    active=$(get_active_environment "$app") || return 1
    case $active in blue) target=green;; green) target=blue;; *) return 2;; esac
    deploy_to_environment "$target" "$app" "$version" || return 1
    check_environment_health "$target" "$app" || return 1
    run_smoke_tests "$target" "$app" || return 1
    switch_traffic "$app" "$target" || return 1
    if check_environment_health "$target" "$app" && update_active_environment "$app" "$target"; then
        printf 'Active environment: %s\n' "$target"
    else
        switch_traffic "$app" "$active" || return 3
        return 1
    fi
}

Canary - Implementation

canary_values_healthy() {
    python3 - "$@" <<'PY'
import math,sys
try:
    error,latency,error_limit,latency_limit=map(float,sys.argv[1:])
    ok=all(map(math.isfinite,(error,latency,error_limit,latency_limit)))
    ok=ok and 0<=error<=100 and 0<=error_limit<=100 and latency>=0 and latency_limit>=0
    raise SystemExit(0 if ok and error<=error_limit and latency<=latency_limit else 1)
except (ValueError,TypeError): raise SystemExit(1)
PY
}
deploy_canary() {
    local app=$1 version=$2 percentage error latency
    local observe=${CANARY_OBSERVE_SECONDS:-60}
    [[ $observe =~ ^[0-9]{1,4}$ && -n ${CANARY_ERROR_THRESHOLD:-} && -n ${CANARY_LATENCY_THRESHOLD:-} ]] || return 2
    canary_values_healthy 0 0 "$CANARY_ERROR_THRESHOLD" "$CANARY_LATENCY_THRESHOLD" || return 2
    deploy_to_canary_pool "$app" "$version" || return 1
    for percentage in 5 25 50 75 100; do
        if set_canary_weight "$app" "$percentage" && monitor_canary_metrics "$app" "$observe" &&
           error=$(get_canary_error_rate "$app") && latency=$(get_canary_latency "$app" p99) &&
           canary_values_healthy "$error" "$latency" "$CANARY_ERROR_THRESHOLD" "$CANARY_LATENCY_THRESHOLD"; then :
        else
            rollback_canary "$app" || return 3
            return 1
        fi
    done
    promote_canary "$app" "$version"
}

Health Checks

check_health() {
    local instance=$1 type=$2 endpoint=$3 duration=${4:-10} status host port
    [[ $duration =~ ^[1-9][0-9]{0,3}$ ]] || return 2
    case $type in
        http)
            status=$(curl --silent --show-error --output /dev/null --write-out '%{http_code}' \
                --connect-timeout "$duration" --max-time "$duration" -- "http://${instance}${endpoint}") || return 1
            [[ $status == 200 ]] ;;
        tcp)
            IFS=: read -r host port <<< "$instance"
            [[ $host =~ ^[A-Za-z0-9._-]+$ && $port =~ ^[1-9][0-9]{0,4}$ ]] || return 2
            ((port <= 65535)) || return 2
            timeout "$duration" bash -c 'exec 3<>"/dev/tcp/$1/$2"' _ "$host" "$port" ;;
        process) process_probe "$instance" "$endpoint" "$duration" ;;
        command) command_probe "$instance" "$endpoint" "$duration" ;;
        *) return 2 ;;
    esac
}
wait_for_health() {
    local instance=$1 duration=${2:-60} interval=${3:-5} deadline remaining attempt
    [[ $duration =~ ^[1-9][0-9]{0,3}$ && $interval =~ ^[1-9][0-9]{0,3}$ ]] || return 2
    deadline=$((SECONDS + duration))
    while ((SECONDS < deadline)); do
        remaining=$((deadline - SECONDS)); attempt=$remaining
        ((attempt <= 10)) || attempt=10
        if check_health "$instance" http /health "$attempt"; then return 0; fi
        remaining=$((deadline - SECONDS))
        ((remaining > 0)) || break
        ((remaining >= interval)) && remaining=$interval
        sleep "$remaining" || return 1
    done
    return 1
}
# process_probe/command_probe are reviewed adapters. No SSH code is built from data.
# The HTTP deadline covers the entire request, not just connection establishment.

The Hooks System

# Hook configuration (app.conf)
declare -A HOOKS=(
    [pre_deploy]="hooks/pre_deploy.sh"
    [build]="hooks/build.sh"
    [post_deploy]="hooks/post_deploy.sh"
    [pre_rollback]="hooks/pre_rollback.sh"
    [post_rollback]="hooks/post_rollback.sh"
    [on_failure]="hooks/on_failure.sh"
)

run_hook() {
    local hook_name="$1"
    shift
    local hook_script="${HOOKS[$hook_name]:-}"
    
    [[ -z "$hook_script" ]] && return 0  # Hook is not configured
    [[ ! -x "$hook_script" ]] && { log_error "Configured hook not executable: $hook_name"; return 1; }
    
    log_info "Running hook: $hook_name"
    
    # Execution with environment and arguments
    if APP="$APP_NAME" VERSION="$VERSION" ENVIRONMENT="$ENVIRONMENT" \
       "$hook_script" "$@"; then
        log_info "✓ Hook $hook_name completed"
        return 0
    else
        local status=$?
        log_error "✗ Hook $hook_name failed with status: $status"
        return $status
    fi
}

Automatic Rollback

rollback() {
    local app=$1 version=${2:-previous} previous_env
    if [[ $version == previous ]]; then version=$(get_previous_version "$app") || return 1; fi
    [[ -n $version ]] || return 1
    run_hook pre_rollback "$version" || return 1
    case $DEPLOY_STRATEGY in
        blue-green)
            previous_env=$(get_previous_environment "$app") || return 1
            [[ $previous_env == blue || $previous_env == green ]] || return 2
            switch_traffic "$app" "$previous_env" || return 1 ;;
        rolling|canary) deploy_rolling "$app" "$version" "${INSTANCES[@]}" || return 1 ;;
        *) return 2 ;;
    esac
    if ! verify_deployment "$app" "$version"; then
        run_hook on_failure rollback_failed || :
        return 1
    fi
    run_hook post_rollback "$version" || return 1
    record_rollback "$app" "$version" || return 1
    printf 'Rollback verified: %s %s\n' "$app" "$version"
}
# Version reversal does not undo persistent data migrations.

Release Management

# From the repository root, using an owned absolute RELEASES_DIR.
RELEASE_HELPER="$PWD/SEM06/scripts/examples/release_reference.py"
create_release() {
    local app=$1 version=$2 artifact=$3 digest=${4:?Expected SHA-256 required}
    python3 "$RELEASE_HELPER" create "$RELEASES_DIR" "$app" "$version" "$artifact" "$digest"
}
activate_release() {
    python3 "$RELEASE_HELPER" activate "$RELEASES_DIR" "$1" "$2"
}
# The maintained helper validates the expected digest and every member path/type,
# refuses an existing version, extracts to an owned stage, writes JSON metadata,
# and publishes only a complete release. Links/devices are explicitly unsupported.
# Activation records previous/target and atomically replaces one current symlink.
# This boundary assumes a trusted quiescent parent and one filesystem.
# Publisher authentication, service health and database changes require separate contracts.

Exercise 2: Multi-Environment

Objective: Deployment to staging then production

deploy_pipeline() {
    local app="$1"
    local version="$2"
    
    # 1. Deploy to STAGING
    log_info "Deploying to STAGING..."
    ENVIRONMENT="staging" deploy_rolling "$app" "$version" "${STAGING_INSTANCES[@]}" || return 1
    
    # 2. Automated tests in staging
    run_integration_tests "$app" "staging" || {
        log_error "Staging tests failed"
        return 1
    }
    
    # 3. Manual approval (if required by an explicitly adopted workflow)
    if [[ "$REQUIRE_APPROVAL" == "true" ]]; then
        read -p "Approve production deployment? (yes/no): " approval
        [[ "$approval" == "yes" ]] || return 3  # cancelled, not deployed
    fi
    
    # 4. Deploy to PRODUCTION
    log_info "Deploying to PRODUCTION..."
    ENVIRONMENT="production" deploy_canary "$app" "$version"
}

# Application pseudocode remains an exercise:
# if feature_enabled("new_checkout", user_id): return new_checkout_flow()
# else: return old_checkout_flow()
# set_feature_flag and rollback_feature_flag are explicit storage adapters.
deploy_with_feature_flag() {
    local app=$1 version=$2 feature=$3 pct delay=${FEATURE_OBSERVE_SECONDS:-300}
    [[ $delay =~ ^[0-9]{1,4}$ ]] || return 2
    set_feature_flag "$feature" 0 "$app" || return 1
    deploy_rolling "$app" "$version" "${INSTANCES[@]}" || return 1
    for pct in 5 25 50 100; do
        if set_feature_flag "$feature" "$pct" "$app" && sleep "$delay" && check_metrics "$app" "$feature"; then :
        else
            rollback_feature_flag "$feature" "$app" || return 3
            return 1
        fi
    done
}

Best Practices

Pre-Deploy

  • Backward-compatible migrations with a tested recovery plan
  • Config validation
  • Backup critical data
  • Notify stakeholders

During Deploy

  • Monitor metrics closely
  • Keep rollback ready
  • Gradual traffic shift
  • Log useful evidence without credentials or unnecessary personal data

Post-Deploy

Smoke tests | Performance baseline | Cleanup old releases | Update documentation
