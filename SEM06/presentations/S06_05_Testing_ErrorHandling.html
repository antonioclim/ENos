Basic Assertions

ASSERTIONS_PASSED=0
ASSERTIONS_FAILED=0
assert_result() {
    local truth=$1 message=$2
    if [[ $truth == 0 ]]; then
        ASSERTIONS_PASSED=$((ASSERTIONS_PASSED + 1))
        printf '  PASS: %s\n' "$message"
        return 0
    fi
    ASSERTIONS_FAILED=$((ASSERTIONS_FAILED + 1))
    printf '  FAIL: %s\n' "$message" >&2
    return 1
}
assert_equals() {
    if [[ $1 == "$2" ]]; then assert_result 0 "${3:-Equal values}";
    else assert_result 1 "${3:-Expected [$1], got [$2]}"; fi
}
assert_contains() {
    if [[ $1 == *"$2"* ]]; then assert_result 0 "${3:-Substring present}";
    else assert_result 1 "${3:-Missing substring [$2]}"; fi
}
assert_success() { assert_equals 0 "$1" "${2:-Command succeeded}"; }

File Assertions

assert_file_exists() {
    if [[ -f $1 ]]; then assert_result 0 "${2:-File exists}";
    else assert_result 1 "${2:-Missing file [$1]}"; fi
}
assert_file_contains() {
    if grep -Fq -- "$2" "$1"; then assert_result 0 "${3:-File contains text}";
    else assert_result 1 "${3:-File lacks text or cannot be read}"; fi
}
assert_file_permissions() {
    local actual
    if actual=$(stat -c %a -- "$1"); then assert_equals "$2" "$actual";
    else assert_result 1 "Cannot stat [$1]"; fi
}
assert_directory_not_empty() {
    local output
    if [[ -d $1 ]] && output=$(find "$1" -mindepth 1 -maxdepth 1 -printf x -quit) && [[ $output == x ]]; then
        assert_result 0 'Directory is nonempty'
    else assert_result 1 'Directory is empty or unavailable'; fi
}

Numeric Assertions

assert_number() {
    local operation=$1 value=$2 lower=$3 upper=${4:-0}
    if python3 - "$operation" "$value" "$lower" "$upper" <<'PY'
import math, sys
try:
    value, lower, upper = map(float, sys.argv[2:])
    valid = all(map(math.isfinite, (value, lower, upper)))
    result = {'gt': value > lower, 'lt': value < lower,
              'range': lower <= value <= upper}[sys.argv[1]]
    raise SystemExit(0 if valid and result else 1)
except (ValueError, KeyError):
    raise SystemExit(1)
PY
    then assert_result 0 "Numeric $operation assertion"
    else assert_result 1 "Numeric $operation assertion ($value)"; fi
}
assert_greater_than() { assert_number gt "$1" "$2"; }
assert_less_than() { assert_number lt "$1" "$2"; }
assert_in_range() { assert_number range "$1" "$2" "$3"; }
# Decimal, finite values are accepted. NaN/infinity and malformed values fail.
assert_in_range 23.5 0 100

# Scope the mock to a child shell; the caller's functions/PATH remain intact.
# From repository root, in a disposable test:
(
    fixture=$(mktemp -d) || exit 1
    trap 'rm -rf -- "$fixture"' EXIT
    helper="$PWD/SEM06/scripts/examples/mock_curl.py"
    printf '#!/bin/bash\nexec python3 %q "$@"\n' "$helper" > "$fixture/curl"
    chmod u+x "$fixture/curl"
    PATH="$fixture:$PATH"
    unset -f curl 2>/dev/null || :  # only in this child, not the caller
    export MOCK_CURL_EXPECTED_ARGV_JSON='["--fail", "https://fixture.invalid/health"]'
    export MOCK_CURL_BODY='{"status":"unavailable"}'
    export MOCK_CURL_HTTP_STATUS=503 MOCK_CURL_EXIT=0
    if curl --fail https://fixture.invalid/health; then rc=0; else rc=$?; fi
    [[ $rc == 22 ]]  # --fail maps HTTP >=400 to curl status22, not shell status503
)
# The Python fixture rejects unexpected argv; response data is never evaluated.
# It implements only its documented option subset, not the whole curl interface.

Test Runner

#!/usr/bin/env bash
# Each file is sourced in a fresh Bash process. Trusted test code only.
set -uo pipefail
SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd) || exit 2
run_file() {
    source "$SCRIPT_DIR/test_helpers.sh" || return 2
    source "$1" || return 2
    local -a functions=()
    local line name fn before status total=0 passed=0 failed=0
    while IFS= read -r line; do
        name=${line##* }
        [[ $name == test_* ]] && functions+=("$name")
    done < <(declare -F)
    ((${#functions[@]} > 0)) || { echo 'NOT_RUN: no test_ functions' >&2; return 2; }
    for fn in "${functions[@]}"; do
        total=$((total + 1)); before=$ASSERTIONS_FAILED; status=0
        if declare -F setup >/dev/null; then setup || status=1; fi
        if ((status == 0)); then "$fn" || status=1; fi
        if declare -F teardown >/dev/null; then teardown || status=1; fi
        if ((status == 0 && ASSERTIONS_FAILED == before)); then
            passed=$((passed + 1))
        else failed=$((failed + 1)); fi
    done
    printf 'Tests: %s total, %s passed, %s failed; assertions: %s passed, %s failed\n' \
        "$total" "$passed" "$failed" "$ASSERTIONS_PASSED" "$ASSERTIONS_FAILED"
    ((failed == 0))
}
main() {
    if [[ ${1:-} == --file ]]; then
        [[ $# == 2 && -r $2 ]] || return 2
        run_file "$2"; return $?
    fi
    [[ $# -le 1 && -d ${1:-tests} ]] || return 2
    local directory=${1:-tests} plan file rc=0 count=0
    plan=$(mktemp) || return 2
    trap 'rm -f -- "$plan"' RETURN
    find "$directory" -type f -name 'test_*.sh' -print0 > "$plan" || return 2
    while IFS= read -r -d '' file; do
        count=$((count + 1))
        bash "${BASH_SOURCE[0]}" --file "$file" || rc=1
    done < "$plan"
    ((count > 0)) || { echo 'NOT_RUN: no test files' >&2; return 2; }
    return "$rc"
}
if [[ ${BASH_SOURCE[0]} == "$0" ]]; then main "$@"; fi

Strict Mode (set options)

# Independent child-shell probes: an expected error must not abort later examples.
if bash -ec 'false; echo unreachable'; then rc=0; else rc=$?; fi
printf 'Eligible errexit failure: %s\n' "$rc"  # 1
if bash -uc 'unset missing; printf "%s" "$missing"'; then rc=0; else rc=$?; fi
printf 'Nounset failure: %s\n' "$rc"  # nonzero
bash -c 'set -o pipefail; (exit 3) | (exit 7); printf "%s\n" "$?"'  # 7
# pipefail = rightmost nonzero status, not first failure in time.
# -e exceptions: tests, non-final &&/|| commands, negated pipelines, tested functions.
# ( ... ) inherits errexit; $( ... ) normally clears it unless inherit_errexit is set.
# ERR may run without -e. set -E inherits ERR into functions/subshell environments.
# set -T inherits DEBUG/RETURN; EXIT is not an operating-system signal.
# Changing IFS globally is not a substitute for quoting or a defined input grammar.

Trap Handlers

TEMP_DIR=''
cleanup() {
    local rc=$?
    if [[ -n $TEMP_DIR ]]; then rm -rf -- "$TEMP_DIR" || :; fi
    # flock is released by closing its owned descriptor/process exit.
    # Never unlink a shared lock pathname while another process can hold its inode.
    exit "$rc"
}
trap cleanup EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
TEMP_DIR=$(mktemp -d) || exit 1
error_handler() {
    local line=$1 code=$2 i=0
    printf 'Failure at line %s: status %s\n' "$line" "$code" >&2
    while caller "$i"; do i=$((i + 1)); done
    return 0
}
trap 'error_handler "$LINENO" "$?"' ERR
# Keep command contents/credentials out of logs. SIGKILL/power loss bypass traps.

Logging Framework

# Log levels
declare -A LOG_LEVELS=([TRACE]=0 [DEBUG]=1 [INFO]=2 [WARN]=3 [ERROR]=4 [FATAL]=5)
LOG_LEVEL="${LOG_LEVEL:-INFO}"

# Colors
declare -A LOG_COLORS=(
    [TRACE]="\033[0;37m"    # Gray
    [DEBUG]="\033[0;36m"    # Cyan
    [INFO]="\033[0;32m"     # Green
    [WARN]="\033[0;33m"     # Yellow
    [ERROR]="\033[0;31m"    # Red
    [FATAL]="\033[1;31m"    # Bold Red
)
NC="\033[0m"

log() {
    local level="$1"
    shift
    local message="$*"
    
    # Check if should log
    local level_num=${LOG_LEVELS[$level]:-2}
    local current_num=${LOG_LEVELS[$LOG_LEVEL]:-2}
    [[ $level_num -lt $current_num ]] && return 0
    
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    local color="${LOG_COLORS[$level]}"
    
    # Format: [TIMESTAMP] [LEVEL] message
    printf '%b[%s] [%-5s]%b %s\n' "$color" "$timestamp" "$level" "$NC" "$message" >&2 || return 1
    
    # Also write to log file if defined
    if [[ -n "${LOG_FILE:-}" ]]; then
        printf '[%s] [%-5s] %s\n' "$timestamp" "$level" "$message" >> "$LOG_FILE" || return 1
    fi
    return 0
}

# Convenience functions
log_trace() { log "TRACE" "$@"; }
log_debug() { log "DEBUG" "$@"; }
log_info()  { log "INFO"  "$@"; }
log_warn()  { log "WARN"  "$@"; }
log_error() { log "ERROR" "$@"; }
log_fatal() { log "FATAL" "$@"; exit 1; }

Input Validation

# Validation functions

validate_not_empty() {
    local value="$1"
    local name="$2"
    
    if [[ -z "$value" ]]; then
        log_error "Required parameter missing: $name"
        return 1
    fi
}

validate_integer() {
    local value="$1"
    local name="$2"
    
    if [[ ! "$value" =~ ^-?[0-9]+$ ]]; then
        log_error "Invalid integer for $name: $value"
        return 1
    fi
}

validate_file_exists() {
    local file="$1"
    
    if [[ ! -f "$file" ]]; then
        log_error "File not found: $file"
        return 1
    fi
    
    if [[ ! -r "$file" ]]; then
        log_error "File not readable: $file"
        return 1
    fi
}

validate_in_list() {
    local value="$1"
    local name="$2"
    shift 2
    local allowed=("$@")
    
    local valid=false
    for item in "${allowed[@]}"; do
        [[ "$value" == "$item" ]] && valid=true && break
    done
    
    if [[ "$valid" == "false" ]]; then
        log_error "Invalid $name: $value. Allowed: ${allowed[*]}"
        return 1
    fi
}

Retry Pattern

#!/usr/bin/env bash
# External argv command only; total timeout also includes retry delays.
retry() {
    [[ $# -ge 4 ]] || return 2
    local attempts=$1 delay=$2 backoff=$3 attempt=1 rc=1; shift 3
    [[ $attempts =~ ^[1-9][0-9]?$|^100$ && $delay =~ ^[0-9]{1,4}$ && $backoff =~ ^[1-9]$|^10$ ]] || return 2
    delay=$((10#$delay)); ((delay <= 3600)) || return 2
    while ((attempt <= attempts)); do
        if "$@"; then return 0; else rc=$?; fi
        if ((attempt < attempts)); then
            sleep "$delay" || return 1
            delay=$((delay * backoff)); ((delay <= 3600)) || delay=3600
        fi
        attempt=$((attempt + 1))
    done
    return "$rc"
}
retry_with_timeout() {
    [[ $# -ge 5 && $1 =~ ^[1-9][0-9]*([.][0-9]+)?$ ]] || return 2
    local duration=$1 source_path; shift
    source_path=$(realpath -- "${BASH_SOURCE[0]}") || return 2
    timeout --signal=TERM --kill-after=2 "$duration" bash -c \
        'source "$1"; shift; retry "$@"' _ "$source_path" "$@"
}
if [[ ${BASH_SOURCE[0]} == "$0" ]]; then retry "$@"; fi

# Examples use external commands. Bare caller-local functions are unsupported:
# retry_with_timeout 3 2 0 1 /bin/true     # 0
# retry_with_timeout 3 2 0 1 /bin/false    # 1, preserved final status
# retry_with_timeout 1 2 0 1 sleep 10      # 124 (GNU timeout)
# A retried command itself returning124 has the same status; retain attempt logs.
