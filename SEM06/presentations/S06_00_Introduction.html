Robust Error Handling

#!/bin/bash
set -euo pipefail
TEMP_FILE=''
cleanup() {
    local rc=$?
    if [[ -n $TEMP_FILE ]]; then rm -f -- "$TEMP_FILE" || :; fi
    exit "$rc"
}
trap cleanup EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
TEMP_FILE=$(mktemp) || exit 1
error_handler() {
    local line=$1 code=$2 i=0
    printf 'Failure at line %s, status %s\n' "$line" "$code" >&2
    while caller "$i"; do i=$((i + 1)); done
    return 0
}
trap 'error_handler "$LINENO" "$?"' ERR
# ERR has test-context exceptions. EXIT cannot handle SIGKILL or power loss.

Testing Framework

# Run maintained tests; they use disposable paths and explicit fixtures.
python3 -m pytest SEM06/tests/test_runtime_semantics.py -q
bash SEM06/scripts/projects/monitor/tests/test_monitor.sh
# Teaching assertion/runner contract, independent of the application:
source SEM06/scripts/examples/test_helpers.sh
assert_in_range 23.5 0 100
if assert_success 7 'A failed command must fail its assertion'; then
    printf 'Unexpected success\n' >&2; exit 1
fi
# Inject two counter snapshots for CPU deltas, not one boot-to-date snapshot.

# Templates are not installed by this lesson; notification/service use needs its own authorisation.
