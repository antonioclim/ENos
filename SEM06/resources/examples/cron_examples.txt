# CAPSTONE cron syntax examples — do not install this reference as a crontab.
# All absolute paths, accounts, permissions and literal config files need review.
# User crontab: minute hour day-of-month month day-of-week command (no user field).
# /etc/crontab and /etc/cron.d add a username before command.
# Local host timezone applies; DST may skip/repeat wall-clock times.
# Restricted day-of-month AND day-of-week fields are matched with OR in common Vixie cron.
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
MAILTO=""

# Monitor single sample every five minutes: status 2 = active alert, not healthy.
*/5 * * * * /bin/bash /opt/ENos/SEM06/scripts/projects/monitor/monitor.sh --config /etc/capstone/monitor.conf --output json >> /absolute/owned/monitor.jsonl 2>> /absolute/owned/monitor.err
# Thresholds are explicit; there is no --all, --cpu, --format=json or --continuous.
* 9-18 * * 1-5 /bin/bash /opt/ENos/SEM06/scripts/projects/monitor/monitor.sh --config /etc/capstone/monitor.conf --cpu-threshold 80 --mem-threshold 85

# Backup uses options directly, never a 'create' subcommand. Shared config selects
# source/destination/lock/log paths and disables notification in a local fixture.
0 3 * * * /bin/bash /opt/ENos/SEM06/scripts/projects/backup/backup.sh --config /etc/capstone/backup.conf --type daily --incremental
0 2 * * 0 /bin/bash /opt/ENos/SEM06/scripts/projects/backup/backup.sh --config /etc/capstone/backup.conf --type weekly --incremental
0 1 1 * * /bin/bash /opt/ENos/SEM06/scripts/projects/backup/backup.sh --config /etc/capstone/backup.conf --type monthly --incremental
# Retention occurs after a successful backup, preserving retained ancestor chains.
# There is no 'rotate' subcommand. --type full creates a full base and skips rotation.

# Deployer status/health require an explicit configured target, e.g. --service.
# A failed probe is evidence for review, not implicit permission to roll back.
*/10 * * * * /bin/bash /opt/ENos/SEM06/scripts/projects/deployer/deployer.sh --config /etc/capstone/deployer.conf --no-notify status --service example-service
# Real service/container operations are host-dependent and are not executed here.

# Schedule-only examples (replace /absolute/owned/task.sh with a bounded real job):
*/15 * * * * /bin/bash /absolute/owned/task.sh
0 */2 * * * /bin/bash /absolute/owned/task.sh
30 8 * * * /bin/bash /absolute/owned/task.sh
0 9 * * 1-5 /bin/bash /absolute/owned/task.sh
0 10 * * 0,6 /bin/bash /absolute/owned/task.sh
0 0 1,15 * * /bin/bash /absolute/owned/task.sh
0-14 * * * * /bin/bash /absolute/owned/task.sh
@reboot /bin/bash /absolute/owned/task.sh
@yearly /bin/bash /absolute/owned/task.sh
@monthly /bin/bash /absolute/owned/task.sh
@weekly /bin/bash /absolute/owned/task.sh
@daily /bin/bash /absolute/owned/task.sh
@hourly /bin/bash /absolute/owned/task.sh

# Bounded job with external lock distinct from the program's own lock. Never unlink
# a held flock path. Do not put a command string in -c when argv is sufficient.
*/5 * * * * flock -n /absolute/owned/cron-wrapper.lock /bin/bash /absolute/owned/task.sh
0 3 * * * timeout --signal=TERM --kill-after=5 3600 /bin/bash /absolute/owned/task.sh
